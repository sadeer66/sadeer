  function routeLengthPx(route){
    var total=0;
    for(var i=1;i<route.points.length;i++){
      var a=route.points[i-1],b=route.points[i];
      total+=Math.hypot(b.x-a.x,b.y-a.y);
    }
    return total;
  }

  function fittingAt(a,b,c){
    var first=Math.atan2(b.y-a.y,b.x-a.x);
    var second=Math.atan2(c.y-b.y,c.x-b.x);
    var turn=(second-first)*180/Math.PI;
    turn=((turn+540)%360)-180;
    var amount=Math.abs(turn);
    if(amount<.001)return null;
    var base={turn:amount,at:b,a:a,b:b,c:c};
    if(Math.abs(amount-45)<.001)return Object.assign(base,{kind:'45',label:'كوع 45°',shortLabel:'45°'});
    if(Math.abs(amount-90)<.001)return Object.assign(base,{kind:'90',label:'كوع 90°',shortLabel:'90°'});
    return Object.assign(base,{kind:'invalid',label:'كوع مفصول — زاوية غير صالحة',shortLabel:'كوع مفصول'});
  }

  function routeFittings(route){
    if(!route||route.autoElbows===false)return [];
    var fittings=[];
    for(var i=1;i<route.points.length-1;i++){
      var fitting=fittingAt(route.points[i-1],route.points[i],route.points[i+1]);
      if(fitting)fittings.push(fitting);
    }
    return fittings;
  }

  function numberFormat(value,decimals){
    var n=finiteNumber(value,0);
    return n.toLocaleString('ar-IQ',{maximumFractionDigits:decimals,minimumFractionDigits:decimals});
  }

  function formatLength(px){
    var scale=typeof pxPerCm!=='undefined'?finiteNumber(pxPerCm,0):0;
    if(!scale)return 'عاير الخارطة لإظهار الطول';
    var metres=px/scale/100;
    if(metres>=1)return numberFormat(metres,metres<10?2:1)+' م';
    return numberFormat(metres*100,0)+' سم';
  }

  function snapToRouteAngle(from,raw,enabled){
    var point=clampRoutePoint(raw);
    if(!from||!enabled)return point;
    var dx=point.x-from.x,dy=point.y-from.y;
    var length=Math.hypot(dx,dy);
    if(length<1)return point;
    var angle=Math.atan2(dy,dx);
    var snapped=Math.round(angle/(Math.PI/4))*(Math.PI/4);
    return clampRoutePoint({
      x:from.x+Math.cos(snapped)*length,
      y:from.y+Math.sin(snapped)*length
    });
  }

  }

  // Straight spans end at the fitting ports. No sharp polyline is drawn underneath.
  function fittingGeometry(f,width){
    var ix=f.a.x-f.b.x,iy=f.a.y-f.b.y,ox=f.c.x-f.b.x,oy=f.c.y-f.b.y;
    var il=Math.hypot(ix,iy),ol=Math.hypot(ox,oy);
    if(il<1||ol<1)return null;
    ix/=il;iy/=il;ox/=ol;oy/=ol;
    var source=catalogueElbowPorts[f.kind];
    var scale=source?width/source.body:0;
    var inLead=source?Math.hypot(source.sx-source.cx,source.sy-source.cy)*scale:width*1.75;
    var outLead=source?Math.hypot(source.ex-source.cx,source.ey-source.cy)*scale:width*1.75;
    return {ix:ix,iy:iy,ox:ox,oy:oy,lead:Math.max(inLead,outLead),source:source,scale:scale,
      start:{x:f.b.x+ix*inLead,y:f.b.y+iy*inLead},end:{x:f.b.x+ox*outLead,y:f.b.y+oy*outLead}};
  }

  function drawFitting(fitting,width,draft){
    var g=fittingGeometry(fitting,width);if(!g)return;
    var img=images[fitting.kind==='45'?'elbow45':'elbow90'];
    var mask=new Path2D();mask.moveTo(g.start.x,g.start.y);mask.quadraticCurveTo(fitting.b.x,fitting.b.y,g.end.x,g.end.y);
    ctx.save();
    if(outlinePass&&fitting.kind==='invalid'){ctx.restore();return;}
    if(fitting.kind==='invalid'){
      ctx.lineCap='butt';ctx.lineWidth=width*1.10;ctx.strokeStyle='#ef4444';ctx.stroke(mask);
      ctx.restore();drawFittingLabel(fitting,width,draft,g.ix,g.iy,g.ox,g.oy);return;
    }
    if(g.source&&img.complete&&img.naturalWidth){
      var source=g.source;
      var ux=source.sx-source.cx,uy=source.sy-source.cy,vx=source.ex-source.cx,vy=source.ey-source.cy,det=ux*vy-uy*vx;
      var sx=g.start.x-fitting.b.x,sy=g.start.y-fitting.b.y,ex=g.end.x-fitting.b.x,ey=g.end.y-fitting.b.y;
      var ax=(sx*vy-ex*uy)/det,ay=(sy*vy-ey*uy)/det;
      var bx=(ex*ux-sx*vx)/det,by=(ey*ux-sy*vx)/det;
      ctx.transform(ax,ay,bx,by,fitting.b.x-ax*source.cx-bx*source.cy,fitting.b.y-ay*source.cx-by*source.cy);
      if(fitting.kind==='45')ctx.filter='brightness(1.08)';
      if(outlinePass){pipeColors.drawOutline(ctx,img,[0,0],activeOutlineWidth/g.scale);ctx.restore();return;}
      ctx.drawImage(pipeColors.tint(img,activePipeColor),0,0);
    }else{
      if(outlinePass){ctx.lineCap='butt';ctx.lineWidth=width*1.10+2*activeOutlineWidth;ctx.strokeStyle='#202833';ctx.stroke(mask);ctx.restore();return;}
      ctx.lineCap='butt';ctx.lineWidth=width*1.10;ctx.strokeStyle=activePipeColor==='#ffffff'?'#f8f9fa':activePipeColor;ctx.stroke(mask);
    }

    if(state.display.fittingLabels)drawFittingLabel(fitting,width,draft,g.ix,g.iy,g.ox,g.oy);
  }

  var angleLabelObstacles=[],angleLabelBoxes=[];

  function segmentMeetsLabel(obstacle,box){
    // Clip the pipe centreline against the label rectangle expanded by its radius.
    var a=obstacle.a,b=obstacle.b,r=obstacle.radius;
    var lo=0,hi=1,dx=b.x-a.x,dy=b.y-a.y;
    var ps=[-dx,dx,-dy,dy],qs=[a.x-box.minX+r,box.maxX-a.x+r,a.y-box.minY+r,box.maxY-a.y+r];
    for(var i=0;i<4;i++){
      if(Math.abs(ps[i])<1e-10){if(qs[i]<0)return false;continue;}
      var t=qs[i]/ps[i];if(ps[i]<0)lo=Math.max(lo,t);else hi=Math.min(hi,t);
      if(lo>hi)return false;
    }
    return true;
  }

  function chooseAngleLabelPosition(p,nx,ny,width,tw,fs,z){
    var padding=1,hx=tw/2+padding,hy=fs*.7+padding;
    var first=width*.55+fs*.8+1,step=Math.max(1,width*.15);
    for(var ring=0;ring<80;ring++){
      var radius=first+ring*step;
      var angles=[0,15,-15,30,-30,45,-45,60,-60,75,-75];
      for(var i=0;i<angles.length;i++){
        var a=angles[i]*Math.PI/180,dx=nx*Math.cos(a)-ny*Math.sin(a),dy=nx*Math.sin(a)+ny*Math.cos(a);
        var x=p.x+dx*radius,y=p.y+dy*radius,box={minX:x-hx,maxX:x+hx,minY:y-hy,maxY:y+hy};
        if(box.minX<0||box.minY<0||box.maxX>canvas.width||box.maxY>canvas.height)continue;
        if(angleLabelObstacles.some(function(o){return o.a?segmentMeetsLabel(o,box):labelBoxesOverlap(o,box);})||angleLabelBoxes.some(function(o){return labelBoxesOverlap(o,box);}))continue;
        angleLabelBoxes.push(box);return {x:x,y:y};
      }
    }
    // Never cover a piece when the outer side has no free space.
    return null;
  }

  function pipeCouplingPositions(a,b,scale,lengths){
    var length=Math.hypot(b.x-a.x,b.y-a.y),result=[];
    if(!(scale>0))return result;
    var totalCm=length/scale,previous=0,index=0;
    while(previous<totalCm){
      var cut=Math.max(10,Math.min(600,Number(lengths&&lengths[index])||600)),distance=previous+cut;
      if(distance>=totalCm-.001)break;
      result.push({x:a.x+(b.x-a.x)*distance/totalCm,y:a.y+(b.y-a.y)*distance/totalCm,index:index,previousCm:previous,distanceCm:distance,totalCm:totalCm});
      previous=distance;index++;
    }
    return result;
  }
  function routeSpan(route,index){

      }
    });
